# Basic Cammand's
********************
ls       = 	Shows the list of files and folders in the current place   <br>
ls -l    = 	Shows files with details like size, date, and permissions <br>
cd folder_name          = Go inside a  <br>
cd ..	   = Go back (up one folder) <br>
pwd      = 	Shows your current location (path) <br>
mkdir folder_name	      = Creates a new folder <br>
rmdir folder_name       =	Deletes an empty folder <br>
rm file_name            =	Deletes a file <br>
rm -r folder_name       =	Deletes a folder with all files inside it <br>
cp source destination	  = Copies a file or folder <br>
cp -r source_folder destination	    = Copies a folder and its contents <br>
mv old_name new_name	              = Moves or renames a file/folder <br>
touch file_name         =	Creates a new empty file <br>
cat file_name	          = Shows the content of a file <br>
nano file_name          =	Opens a file in a simple editor to edit it <br>
